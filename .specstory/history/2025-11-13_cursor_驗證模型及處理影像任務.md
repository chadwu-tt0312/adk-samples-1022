#!/usr/bin/env python3
"""
驗證 final_solution.py 是否能正確處理 test.csv 並產生預測結果

使用方法：
    python validate_final_solution.py --solution-path workspace/Titanic/ensemble/final_solution.py
    或
    python validate_final_solution.py --solution-path workspace/california-housing-prices/ensemble/final_solution.py
"""

import argparse
import subprocess
import sys
import os
from pathlib import Path
import pandas as pd
import json


def check_file_exists(file_path: Path, description: str) -> bool:
    """檢查檔案是否存在"""
    if not file_path.exists():
        print(f"❌ {description} 不存在：{file_path}")
        return False
    print(f"✅ {description} 存在：{file_path}")
    return True


def check_csv_format(csv_path: Path, description: str) -> tuple[bool, pd.DataFrame | None]:
    """檢查 CSV 檔案格式是否正確"""
    try:
        df = pd.read_csv(csv_path, nrows=5)  # 只讀取前 5 行來檢查格式
        print(f"✅ {description} 格式正確，欄位：{', '.join(df.columns.tolist())}")
        return True, df
    except Exception as e:
        print(f"❌ {description} 格式錯誤：{e}")
        return False, None


def validate_solution_structure(solution_path: Path) -> dict:
    """
    驗證 final_solution.py 的結構
    
    Returns:
        dict: 包含驗證結果的字典
    """
    results = {
        'solution_exists': False,
        'input_dir_exists': False,
        'test_csv_exists': False,
        'train_csv_exists': False,
        'solution_readable': False,
        'has_test_loading': False,
        'has_prediction': False,
        'has_submission_output': False,
        'errors': []
    }
    
    # 檢查 solution 檔案
    if not solution_path.exists():
        results['errors'].append(f"final_solution.py 不存在：{solution_path}")
        return results
    results['solution_exists'] = True
    
    # 檢查 input 目錄
    input_dir = solution_path.parent / 'input'
    if not input_dir.exists():
        results['errors'].append(f"input 目錄不存在：{input_dir}")
        return results
    results['input_dir_exists'] = True
    
    # 檢查 test.csv
    test_csv = input_dir / 'test.csv'
    if not test_csv.exists():
        results['errors'].append(f"test.csv 不存在：{test_csv}")
    else:
        results['test_csv_exists'] = True
    
    # 檢查 train.csv
    train_csv = input_dir / 'train.csv'
    if not train_csv.exists():
        results['errors'].append(f"train.csv 不存在：{train_csv}")
    else:
        results['train_csv_exists'] = True
    
    # 讀取 solution 檔案內容
    try:
        with open(solution_path, 'r', encoding='utf-8') as f:
            content = f.read()
        results['solution_readable'] = True
        
        # 檢查是否包含 test.csv 讀取
        if 'test.csv' in content or "read_csv('./input/test.csv')" in content or 'read_csv("./input/test.csv")' in content:
            results['has_test_loading'] = True
        else:
            results['errors'].append("final_solution.py 中未找到讀取 test.csv 的程式碼")
        
        # 檢查是否包含預測
        if 'predict' in content.lower():
            results['has_prediction'] = True
        else:
            results['errors'].append("final_solution.py 中未找到預測相關的程式碼")
        
        # 檢查是否包含輸出 submission
        if 'submission' in content.lower() or 'final' in content.lower():
            results['has_submission_output'] = True
        else:
            results['errors'].append("final_solution.py 中未找到輸出 submission 的程式碼")
            
    except Exception as e:
        results['errors'].append(f"無法讀取 final_solution.py：{e}")
    
    return results


def run_solution(solution_path: Path, timeout: int = 300) -> dict:
    """
    執行 final_solution.py 並檢查結果
    
    Args:
        solution_path: final_solution.py 的路徑
        timeout: 執行超時時間（秒）
    
    Returns:
        dict: 包含執行結果的字典
    """
    results = {
        'execution_success': False,
        'submission_exists': False,
        'submission_valid': False,
        'model_trained': False,
        'predictions_made': False,
        'errors': [],
        'output': '',
        'submission_path': None
    }
    
    # 切換到 solution 所在的目錄
    work_dir = solution_path.parent
    original_dir = os.getcwd()
    
    try:
        os.chdir(work_dir)
        print(f"\n📂 工作目錄：{work_dir}")
        
        # 執行 final_solution.py
        print(f"\n🚀 執行 final_solution.py...")
        try:
            result = subprocess.run(
                [sys.executable, solution_path.name],
                capture_output=True,
                text=True,
                timeout=timeout,
                cwd=work_dir
            )
            
            results['output'] = result.stdout + result.stderr
            
            if result.returncode == 0:
                results['execution_success'] = True
                print("✅ final_solution.py 執行成功")
            else:
                results['errors'].append(f"執行失敗，返回碼：{result.returncode}")
                print(f"❌ 執行失敗：\n{result.stderr}")
                return results
                
        except subprocess.TimeoutExpired:
            results['errors'].append(f"執行超時（超過 {timeout} 秒）")
            print(f"❌ 執行超時")
            return results
        except Exception as e:
            results['errors'].append(f"執行時發生錯誤：{e}")
            print(f"❌ 執行時發生錯誤：{e}")
            return results
        
        # 檢查輸出目錄和 submission 檔案
        final_dir = work_dir / 'final'
        if final_dir.exists():
            submission_files = list(final_dir.glob('submission.csv'))
            if submission_files:
                results['submission_exists'] = True
                results['submission_path'] = submission_files[0]
                print(f"✅ 找到 submission 檔案：{submission_files[0]}")
                
                # 驗證 submission 檔案格式
                try:
                    submission_df = pd.read_csv(submission_files[0])
                    print(f"✅ submission.csv 格式正確")
                    print(f"   行數：{len(submission_df)}")
                    print(f"   欄位：{', '.join(submission_df.columns.tolist())}")
                    
                    # 檢查是否有預測值
                    if len(submission_df) > 0:
                        results['predictions_made'] = True
                        print(f"✅ 已產生 {len(submission_df)} 筆預測結果")
                        
                        # 顯示前幾筆預測
                        print(f"\n📊 前 5 筆預測結果：")
                        print(submission_df.head().to_string(index=False))
                        
                        results['submission_valid'] = True
                    else:
                        results['errors'].append("submission.csv 為空")
                except Exception as e:
                    results['errors'].append(f"無法讀取 submission.csv：{e}")
            else:
                results['errors'].append(f"在 {final_dir} 中找不到 submission.csv")
        else:
            results['errors'].append(f"final 目錄不存在：{final_dir}")
        
        # 檢查是否有模型訓練的輸出（從 stdout 中尋找）
        if 'accuracy' in results['output'].lower() or 'performance' in results['output'].lower() or 'score' in results['output'].lower():
            results['model_trained'] = True
            print("\n✅ 檢測到模型訓練輸出")
        
    finally:
        os.chdir(original_dir)
    
    return results


def main():
    parser = argparse.ArgumentParser(
        description="驗證 final_solution.py 是否能正確處理 test.csv 並產生預測結果",
        formatter_class=argparse.RawDescriptionHelpFormatter,
        epilog="""
範例：

    )
    
    parser.add_argument(
        '--solution-path',
        type=str,
        required=True,
        help='final_solution.py 的完整路徑'
    )
    
    parser.add_argument(
        '--check-only',
        action='store_true',
        help='只檢查結構，不執行程式碼'
    )
    
    parser.add_argument(
        '--timeout',
        type=int,
        default=300,
        help='執行超時時間（秒），預設為 300'
    )
    
    args = parser.parse_args()
    
    solution_path = Path(args.solution_path)
    if not solution_path.is_absolute():
        # 如果是相對路徑，從專案根目錄開始
        base_dir = Path(__file__).parent
        solution_path = base_dir / solution_path
    
    print("="*70)
    print("🔍 驗證 final_solution.py")
    print("="*70)
    print(f"📄 目標檔案：{solution_path}")
    
    # 步驟 1：檢查結構
    print("\n" + "="*70)
    print("步驟 1：檢查檔案結構")
    print("="*70)
    
    structure_results = validate_solution_structure(solution_path)
    
    if not structure_results['solution_exists']:
        print("\n❌ 驗證失敗：找不到 final_solution.py")
        for error in structure_results['errors']:
            print(f"   - {error}")
        sys.exit(1)
    
    # 顯示檢查結果
    checks = [
        ('solution_exists', 'final_solution.py 存在'),
        ('input_dir_exists', 'input 目錄存在'),
        ('test_csv_exists', 'test.csv 存在'),
        ('train_csv_exists', 'train.csv 存在'),
        ('solution_readable', 'final_solution.py 可讀取'),
        ('has_test_loading', '包含讀取 test.csv 的程式碼'),
        ('has_prediction', '包含預測相關程式碼'),
        ('has_submission_output', '包含輸出 submission 的程式碼'),
    ]
    
    for key, description in checks:
        status = "✅" if structure_results[key] else "❌"
        print(f"{status} {description}")
    
    if structure_results['errors']:
        print("\n⚠️  發現以下問題：")
        for error in structure_results['errors']:
            print(f"   - {error}")
    
    # 如果只檢查結構，就結束
    if args.check_only:
        print("\n" + "="*70)
        print("✅ 結構檢查完成（未執行程式碼）")
        print("="*70)
        sys.exit(0)
    
    # 步驟 2：執行並驗證
    print("\n" + "="*70)
    print("步驟 2：執行 final_solution.py 並驗證結果")
    print("="*70)
    
    execution_results = run_solution(solution_path, timeout=args.timeout)
    
    # 顯示執行結果
    execution_checks = [
        ('execution_success', '執行成功'),
        ('model_trained', '模型已訓練'),
        ('predictions_made', '已產生預測結果'),
        ('submission_exists', 'submission.csv 已產生'),
        ('submission_valid', 'submission.csv 格式正確'),
    ]
    
    for key, description in execution_checks:
        status = "✅" if execution_results[key] else "❌"
        print(f"{status} {description}")
    
    if execution_results['errors']:
        print("\n⚠️  執行時發現以下問題：")
        for error in execution_results['errors']:
            print(f"   - {error}")
    
    # 總結
    print("\n" + "="*70)
    print("📋 驗證總結")
    print("="*70)
    
    all_passed = (
        structure_results['solution_exists'] and
        structure_results['input_dir_exists'] and
        structure_results['test_csv_exists'] and
        structure_results['train_csv_exists'] and
        structure_results['has_test_loading'] and
        structure_results['has_prediction'] and
        structure_results['has_submission_output'] and
        execution_results['execution_success'] and
        execution_results['submission_exists'] and
        execution_results['submission_valid']
    )
    
    if all_passed:
        print("✅ 驗證通過！final_solution.py 可以正確處理 test.csv 並產生預測結果")
        if execution_results['submission_path']:
            print(f"📄 預測結果已儲存至：{execution_results['submission_path']}")
    else:
        print("❌ 驗證未完全通過，請檢查上述問題")
        sys.exit(1)


if __name__ == "__main__":
    main()

## 問題 2：MLE-STAR 可以處理影像型的任務嗎？

### 簡短答案

**目前版本主要專注於表格數據（Tabular Data）任務，未明確支援影像型任務。**

### 詳細說明

#### 1. 目前支援的任務類型

根據文件與程式碼，MLE-STAR 目前主要支援：

- ✅ **表格數據分類任務**（Tabular Classification）
  - 範例：Titanic 生存預測
  - 使用的模型：XGBoost、LightGBM、RandomForest 等

- ✅ **表格數據迴歸任務**（Tabular Regression）
  - 範例：California Housing Prices 房價預測
  - 使用的模型：CatBoost、LightGBM、XGBoost 等

#### 2. 技術限制

從程式碼範例可以看出，MLE-STAR 目前使用的模型都是針對表格數據設計的：

- **LightGBM**：梯度提升決策樹，主要用於表格數據
- **XGBoost**：梯度提升框架，主要用於表格數據
- **CatBoost**：梯度提升框架，主要用於表格數據
- **RandomForest**：隨機森林，主要用於表格數據

這些模型**不適合直接處理影像數據**，因為：
- 影像數據是 2D/3D 陣列（像素值）
- 需要卷積神經網路（CNN）等深度學習模型
- 需要不同的資料預處理流程（影像增強、正規化等）

#### 3. 理論上的可能性

雖然目前版本未明確支援，但從架構來看，MLE-STAR **理論上可能可以擴展**到影像任務，因為：

1. **架構彈性**：
   - MLE-STAR 使用 LLM 搜尋最佳實踐並生成程式碼
   - 如果 LLM 能找到影像分類的範例程式碼（如使用 ResNet、EfficientNet 等 CNN 模型），理論上可以生成對應的程式碼

2. **自動化流程**：
   - 資料清理、特徵工程、模型選擇、超參數調整等流程在影像任務中同樣適用
   - 只是具體的實作方式不同

3. **需要調整的部分**：
   - 資料載入方式（從 CSV 改為影像檔案或影像資料集）
   - 模型選擇（從樹模型改為 CNN 模型）
   - 特徵工程（從表格特徵改為影像增強、資料擴增等）

#### 4. 實際建議

如果您需要處理影像型任務，建議：

1. **檢查最新版本**：
   - 查看 MLE-STAR 的最新文件或 GitHub Issues，確認是否有影像任務的支援

2. **嘗試自訂任務**：
   - 可以嘗試在 `tasks/` 目錄下建立影像任務的資料夾
   - 提供 `task_description.txt` 描述影像分類任務
   - 但需要確保資料格式符合系統預期

3. **使用其他工具**：
   - 對於影像任務，可以考慮使用專門的 AutoML 工具，如：
     - AutoKeras
     - Google Cloud AutoML Vision
     - H2O.ai

### 總結

| 項目 | 狀態 |
|------|------|
| 表格數據分類 | ✅ 支援 |
| 表格數據迴歸 | ✅ 支援 |
| 影像分類 | ❓ 未明確支援，可能需要自訂 |
| 影像物件偵測 | ❌ 不支援 |

## 問題 1.5：final_solution.py 只輸入 test.csv，不輸入 train.csv，能得到模型及預測結果嗎？

### 簡短答案

**❌ 不行。`train.csv` 是絕對必要的，無法只使用 `test.csv` 來訓練模型和產生預測結果。**

### 詳細說明

從 `final_solution.py` 的程式碼分析可以看出，`train.csv` 在整個流程中扮演多個關鍵角色：

#### 1. **訓練模型（最關鍵）**

```10:40:machine_learning_engineering/workspace/Titanic/ensemble/final_solution.py
train_df = pd.read_csv('./input/train.csv')
# ... 資料預處理 ...
X = train_df.drop('Survived', axis=1)
y = train_df['Survived']
```

- **第 10 行**：必須讀取 `train.csv`
- **第 38-40 行**：從 `train.csv` 提取特徵（X）和目標值（y）
- **第 72 行**：使用 X 和 y 訓練模型 `final_xgb_clf.fit(X, y)`

**沒有 `train.csv`，就沒有標記數據（labeled data），無法訓練模型。**

#### 2. **計算預處理統計值**

```12:18:machine_learning_engineering/workspace/Titanic/ensemble/final_solution.py
# --- Store imputation values from the training data for consistent preprocessing of test data ---
# Calculate median age from training data
median_age = train_df['Age'].median()
# Calculate mode embarked from training data
mode_embarked = train_df['Embarked'].mode()[0]
# Calculate median fare from training data (test data is known to have missing Fares)
median_fare = train_df['Fare'].median()
```

這些統計值用於處理 `test.csv` 的缺失值：

```84:91:machine_learning_engineering/workspace/Titanic/ensemble/final_solution.py
# Fill missing Age values with the median from the training data
test_df['Age'].fillna(median_age, inplace=True)

# Fill missing Embarked values with the mode from the training data
test_df['Embarked'].fillna(mode_embarked, inplace=True)

# Fill missing Fare values with the median from the training data (as test data has missing Fares)
test_df['Fare'].fillna(median_fare, inplace=True)
```

**為什麼要用訓練數據的統計值？**
- 避免資料洩漏（Data Leakage）：如果使用測試數據的統計值，會讓模型「看到」測試數據的資訊
- 確保一致性：訓練和測試使用相同的預處理參數

#### 3. **確定特徵欄位和順序**

```99:110:machine_learning_engineering/workspace/Titanic/ensemble/final_solution.py
# Align columns between training and test sets to ensure consistency
# This handles potential discrepancies in one-hot encoded columns if
# not all categories are present in both train and test.
missing_cols_in_test = set(X.columns) - set(test_df.columns)
for c in missing_cols_in_test:
    test_df[c] = 0 # Add missing columns to test_df and fill with 0

extra_cols_in_test = set(test_df.columns) - set(X.columns)
if len(extra_cols_in_test) > 0:
    test_df.drop(columns=list(extra_cols_in_test), inplace=True) # Drop extra columns from test_df

test_df = test_df[X.columns] # Ensure the order of columns is the same as in training data
```

**第 110 行**：使用訓練數據的欄位順序 `X.columns` 來對齊測試數據。

### 機器學習的基本原理

這反映了機器學習的基本原理：

1. **監督式學習（Supervised Learning）**需要：
   - **訓練數據（Training Data）**：包含特徵和正確答案（標籤）
   - **測試數據（Test Data）**：只包含特徵，需要預測答案

2. **訓練階段**：
   - 模型從訓練數據中學習「特徵 → 答案」的對應關係
   - 沒有訓練數據，模型無法學習

3. **預測階段**：
   - 使用訓練好的模型對測試數據進行預測
   - 沒有訓練好的模型，無法進行預測

| `test.csv` | 進行預測、產生 submission | ✅ **必要** |
