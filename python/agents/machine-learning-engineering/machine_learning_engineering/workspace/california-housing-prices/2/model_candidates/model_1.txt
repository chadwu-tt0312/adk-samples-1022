## Model name
LightGBM Regressor

## Example Python code
import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.metrics import mean_squared_error
import lightgbm as lgb

# Load the datasets (assuming train.csv and test.csv are available)
# For demonstration, creating dummy data similar to the problem description
# In a real scenario, you would load your actual train.csv and test.csv
try:
    train_df = pd.read_csv('train.csv')
    test_df = pd.read_csv('test.csv')
except FileNotFoundError:
    print("Creating dummy data as 'train.csv' or 'test.csv' not found.")
    np.random.seed(42)
    num_train_samples = 1000
    num_test_samples = 300
    num_features = 10
    
    # Generate random features
    X_dummy = pd.DataFrame(np.random.rand(num_train_samples + num_test_samples, num_features), 
                           columns=[f'feature_{i}' for i in range(num_features)])
    # Generate a target variable (median_house_value) with some noise
    y_dummy = pd.Series(50000 + 200000 * np.random.rand(num_train_samples + num_test_samples) + 
                        np.sum(X_dummy * [10000, 5000, 2000, 1000, 500, 200, 100, 50, 20, 10], axis=1) + 
                        np.random.randn(num_train_samples + num_test_samples) * 10000)

    train_df = pd.concat([X_dummy.iloc[:num_train_samples], y_dummy.iloc[:num_train_samples].rename('median_house_value')], axis=1)
    test_df = X_dummy.iloc[num_train_samples:].copy()

# Separate features and target from the training data
X_train = train_df.drop('median_house_value', axis=1)
y_train = train_df['median_house_value']

# Assume test_df contains only features for prediction
X_test = test_df.copy()

# For simplicity, we'll use a basic train-test split for validation
# In a competition, you'd typically use all train_df for training or cross-validation
X_train_split, X_val_split, y_train_split, y_val_split = train_test_split(X_train, y_train, test_size=0.2, random_state=42)

# Initialize LightGBM Regressor
lgb_model = lgb.LGBMRegressor(objective='regression_l1', # MAE is often robust, 'regression_l2' for MSE
                            metric='rmse',
                            n_estimators=1000,
                            learning_rate=0.05,
                            num_leaves=31,
                            max_depth=-1,
                            random_state=42,
                            n_jobs=-1)

# Train the model
lgb_model.fit(X_train_split, y_train_split,
              eval_set=[(X_val_split, y_val_split)],
              eval_metric='rmse',
              callbacks=[lgb.early_stopping(100, verbose=False)]) # Early stopping to prevent overfitting

# Make predictions on the test set
predictions_lgb = lgb_model.predict(X_test)

# Optional: Evaluate on the validation set if you have labels
val_predictions_lgb = lgb_model.predict(X_val_split)
rmse_lgb = np.sqrt(mean_squared_error(y_val_split, val_predictions_lgb))
print(f"LightGBM Validation RMSE: {rmse_lgb:.4f}")

# Create submission file (example format)
submission_df_lgb = pd.DataFrame({'median_house_value': predictions_lgb})
# submission_df_lgb.to_csv('submission_lgb.csv', index=False)
print("LightGBM predictions generated.")
